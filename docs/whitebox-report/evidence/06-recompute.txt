=== ws06.py ===
"""Recompute the /api/v1/analytics/projection formulas by hand from the raw
Elasticsearch history in docs/whitebox-report/evidence/06-history.json, and
compare against docs/whitebox-report/evidence/06-projection.json.

Formulas copied from api/app/api/analytics.py:486-565 (see plan Task 8 /
docs/whitebox-report/06-quality-forecast.md part 5 for citations).
"""
import json
import math
from datetime import datetime

with open("06-history.json", encoding="utf-8") as f:
    history = json.load(f)

hits = history["hits"]["hits"]
print(f"total hits in index for this table (from 'total'): {history['hits']['total']['value']}")
print(f"hits returned (size=10, sort desc): {len(hits)}")

# The API sorts desc and then reverses to ascending (oldest -> newest):
# scores = [hit["_source"]["quality_score"] for hit in hits][::-1]
scores = [h["_source"]["quality_score"] for h in hits][::-1]
timestamps = [h["_source"]["timestamp"] for h in hits][::-1]

print("\nscores (oldest -> newest):", scores)
print("timestamps (oldest -> newest):")
for t in timestamps:
    print(" ", t)

# x = days since first run (analytics.py:488-493)
t0 = datetime.fromisoformat(timestamps[0].replace("Z", "+00:00")).replace(tzinfo=None)
x = []
for ts in timestamps:
    t = datetime.fromisoformat(ts.replace("Z", "+00:00")).replace(tzinfo=None)
    days_diff = (t - t0).total_seconds() / 86400.0
    x.append(days_diff)

# analytics.py:496-497 fallback to index if all timestamps collapse to day 0
if x[-1] == 0:
    print("\nx[-1] == 0 -> falling back to index-based x (analytics.py:496-497)")
    x = list(range(len(scores)))

print("\nx (days since first run):")
for xi in x:
    print(f"  {xi:.6f}")

y = scores
n = len(scores)
sum_x = sum(x)
sum_y = sum(y)
sum_xx = sum(xi * xi for xi in x)
sum_xy = sum(xi * yi for xi, yi in zip(x, y))
denom = (n * sum_xx - sum_x * sum_x)

print(f"\nn = {n}")
print(f"sum_x = {sum_x}")
print(f"sum_y = {sum_y}")
print(f"sum_xx = {sum_xx}")
print(f"sum_xy = {sum_xy}")
print(f"denom = n*sum_xx - sum_x^2 = {denom}")

if denom != 0:
    m = (n * sum_xy - sum_x * sum_y) / denom
else:
    m = -0.5
    print("denom == 0 -> m fixed at -0.5 (analytics.py:506)")

c = (sum_y - m * sum_x) / n

print(f"\nOLS slope m = {m}")
print(f"OLS intercept c = {c}")

# Standard error of the regression (analytics.py:509-513)
sse = sum((y[i] - (m * x[i] + c)) ** 2 for i in range(n))
print(f"\nSSE = sum((y_i - (m*x_i + c))^2) = {sse}")
if n > 2:
    variance = sse / (n - 2)
else:
    variance = 2.0
    print("n <= 2 -> variance fixed at 2.0 (analytics.py:511)")
print(f"variance = SSE/(n-2) = {variance}")
std_error = variance ** 0.5
if std_error < 0.5:
    print(f"std_error {std_error} < 0.5 floor -> clamped to 0.5 (analytics.py:513)")
    std_error = 0.5
print(f"std_error (after floor) = {std_error}")

# Projections for day 1..7 (analytics.py:515-529)
projected_scores = []
ci_high = []
ci_low = []
last_day = x[-1]
print(f"\nlast_day (x[-1]) = {last_day}")
for day_offset in range(1, 8):
    future_day = last_day + day_offset
    raw_val = c + (future_day * m)
    proj_val = max(0.0, min(100.0, raw_val))
    proj_val_r = round(proj_val, 2)
    projected_scores.append(proj_val_r)

    margin = std_error * (1 + (day_offset * 0.2))
    high = round(min(100.0, proj_val + margin), 2)
    low = round(max(0.0, proj_val - margin), 2)
    ci_high.append(high)
    ci_low.append(low)
    print(f"day {day_offset}: future_day={future_day:.4f} raw={raw_val:.4f} "
          f"clamped={proj_val_r} margin={margin:.4f} ci_high={high} ci_low={low}")

print(f"\nprojected_scores = {projected_scores}")
print(f"ci_high = {ci_high}")
print(f"ci_low  = {ci_low}")

# Stability index (analytics.py:531-535)
mean_score = sum(scores) / len(scores)
variance2 = sum((s - mean_score) ** 2 for s in scores) / len(scores)
std_dev = variance2 ** 0.5
stability = max(5.0, min(100.0, 100.0 - (std_dev * 4.0)))
print(f"\nmean_score = {mean_score}")
print(f"population variance of scores = {variance2}")
print(f"std_dev (sigma) = {std_dev}")
print(f"stability_index = clamp(100 - 4*sigma, 5, 100) = {stability}")

# SLA breach probability (analytics.py:537-550)
sla_threshold = 95.0
if scores[-1] < sla_threshold:
    breach_prob = 99.9
    print(f"\nlast score {scores[-1]} < 95 -> breach_prob fixed at 99.9 (analytics.py:540-541)")
else:
    max_breach = 0.0
    for val in projected_scores:
        z_sla = (val - sla_threshold) / std_error
        prob = 0.5 * (1.0 - math.erf(z_sla / math.sqrt(2.0)))
        print(f"  proj={val} z=(proj-95)/se={z_sla:.6f} "
              f"prob=0.5*(1-erf(z/sqrt2))={prob * 100:.6f}%")
        if prob > max_breach:
            max_breach = prob
    breach_prob = max(0.1, min(99.9, max_breach * 100.0))
print(f"sla_breach_probability = {breach_prob:.1f}%")

# Trend description (analytics.py:552-553)
trend_desc = "Decline detected" if m < 0 else "Stable or improving trend detected"
trend_desc += f" in pipeline runs (slope: {m:.3f} per run)"
print(f"\nhistorical_trend = {trend_desc!r}")

# Crisis forecast (analytics.py:555-565)
days_until_crisis = 7
crisis_component = "None"
crisis_reason = "No quality crisis predicted in the next 7 days."
severity = "LOW"
for idx, val in enumerate(projected_scores):
    if val < 90.0:
        days_until_crisis = idx + 1
        crisis_component = "Ingestion Pipeline Gateway" if m < -1.0 else "Data Quality Validation Layer"
        crisis_reason = (f"Quality score is projected to drop below 90% "
                          f"(estimated: {val}%) due to cumulative errors.")
        severity = "CRITICAL" if val < 80.0 else "WARNING"
        break
print(f"crisis_forecast = days_until_crisis={days_until_crisis}, "
      f"impacted_component={crisis_component!r}, severity={severity!r}")
print(f"crisis_reason = {crisis_reason!r}")

# ---- Compare against the recorded API response ----
with open("06-projection.json", encoding="utf-8") as f:
    api_result = json.load(f)

print("\n=== COMPARISON WITH docs/whitebox-report/evidence/06-projection.json ===")


def close(a, b, tol=0.01):
    return abs(a - b) <= tol


ok = True
for i, (mine, theirs) in enumerate(zip(projected_scores, api_result["projected_scores"])):
    match = close(mine, theirs)
    ok &= match
    print(f"projected_scores[{i}]: mine={mine} api={theirs} match={match}")
for i, (mine, theirs) in enumerate(zip(ci_high, api_result["ci_high"])):
    match = close(mine, theirs)
    ok &= match
    print(f"ci_high[{i}]: mine={mine} api={theirs} match={match}")
for i, (mine, theirs) in enumerate(zip(ci_low, api_result["ci_low"])):
    match = close(mine, theirs)
    ok &= match
    print(f"ci_low[{i}]: mine={mine} api={theirs} match={match}")

my_stability_str = f"{stability:.1f}%"
print(f"stability_index: mine={my_stability_str} api={api_result['stability_index']} "
      f"match={my_stability_str == api_result['stability_index']}")
ok &= my_stability_str == api_result['stability_index']

my_breach_str = f"{breach_prob:.1f}%"
print(f"sla_breach_probability: mine={my_breach_str} api={api_result['sla_breach_probability']} "
      f"match={my_breach_str == api_result['sla_breach_probability']}")
ok &= my_breach_str == api_result['sla_breach_probability']

print(f"historical_trend: mine={trend_desc!r} api={api_result['historical_trend']!r} "
      f"match={trend_desc == api_result['historical_trend']}")
ok &= trend_desc == api_result['historical_trend']

print(f"crisis_forecast.severity: mine={severity!r} api={api_result['crisis_forecast']['severity']!r} "
      f"match={severity == api_result['crisis_forecast']['severity']}")
ok &= severity == api_result['crisis_forecast']['severity']

print(f"\n=== OVERALL MATCH (within 0.01 / exact string): {ok} ===")
